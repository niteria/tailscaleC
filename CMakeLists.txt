cmake_minimum_required(VERSION 3.20)
project(tailscalec VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)

set(TSC_CONFIG_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/config.h" CACHE FILEPATH
    "Shared feature configuration header")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(TSC_STANDALONE ON)
else()
    set(TSC_STANDALONE OFF)
endif()
option(TSC_BUILD_EXAMPLES "Build embedding examples" ${TSC_STANDALONE})
option(TSC_BUILD_TESTS "Build library tests" ${TSC_STANDALONE})
option(TSC_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

if(NOT UNIX)
    message(FATAL_ERROR "This build uses the POSIX adapter. Use ESP-IDF for ESP32.")
endif()

include(cmake/features.cmake)
tsc_read_features(TSC)
include(cmake/library.cmake)
tsc_collect_sources(TSC_SOURCES TSC_PRIVATE_INCLUDES)
list(APPEND TSC_SOURCES platform/posix/system.c)
set(TSC_NEEDS_CRYPTO OFF)
set(TSC_NEEDS_STREAM OFF)
if(TSC_CLIENT OR TSC_DISCO OR TSC_DERP OR TSC_TLS)
    set(TSC_NEEDS_CRYPTO ON)
    list(APPEND TSC_SOURCES platform/posix/crypto.c)
endif()
if(TSC_CLIENT OR TSC_TLS)
    set(TSC_NEEDS_STREAM ON)
    list(APPEND TSC_SOURCES platform/posix/stream.c)
endif()
add_library(tailscalec STATIC ${TSC_SOURCES})
add_library(tailscalec::tailscalec ALIAS tailscalec)
target_include_directories(tailscalec PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
target_include_directories(tailscalec PRIVATE ${TSC_PRIVATE_INCLUDES})
include(cmake/config.cmake)
tsc_configure_target(tailscalec)
if(TSC_NEEDS_CRYPTO)
    find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
    target_link_libraries(tailscalec PUBLIC OpenSSL::Crypto)
endif()

# Install the exact configuration used to compile this archive. A consumer of
# this package must not silently select a different public structure layout.
include(CMakePackageConfigHelpers)
get_target_property(TSC_BUILT_CONFIG tailscalec TSC_CONFIG_HEADER)
configure_file("${TSC_BUILT_CONFIG}"
    "${CMAKE_CURRENT_BINARY_DIR}/generated/tailscalec/build_config.h" COPYONLY)
configure_package_config_file(cmake/tailscalecConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/tailscalecConfig.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/tailscalec")
write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/tailscalecConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(TARGETS tailscalec EXPORT tailscalecTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/tailscalec DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/generated/tailscalec/build_config.h"
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/tailscalec")
install(EXPORT tailscalecTargets NAMESPACE tailscalec::
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/tailscalec")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/tailscalecConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/tailscalecConfigVersion.cmake"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/tailscalec")
install(FILES LICENSE THIRD_PARTY_NOTICES.md
    DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/tailscalec")
install(FILES third_party/wireguard/LICENSE third_party/wireguard/LICENSE.RFC7693
    DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/tailscalec/wireguard")
install(FILES src/discovery/LICENSE.tailtiny
    DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/tailscalec")
install(FILES src/tcp/lwip/LICENSE
    DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/tailscalec/lwip")
if(TSC_TLS)
    find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL)
    target_link_libraries(tailscalec PUBLIC OpenSSL::SSL)
endif()

if(TSC_SANITIZERS)
    target_compile_options(tailscalec PUBLIC -fsanitize=address,undefined
        -fno-omit-frame-pointer)
    target_link_options(tailscalec PUBLIC -fsanitize=address,undefined)
endif()

if(TSC_APP_TCP)
    include(src/tcp/tcp.cmake)
    tsc_add_tcp(tailscalec)
endif()

if(TSC_BUILD_EXAMPLES)
    add_executable(memory-report examples/posix/memory_report.c)
    target_link_libraries(memory-report PRIVATE tailscalec::tailscalec)
    add_executable(stun-probe examples/posix/stun_probe.c)
    target_link_libraries(stun-probe PRIVATE tailscalec::tailscalec)
    if(TSC_CLIENT)
        if(TSC_APP_UDP)
            add_executable(tailnet-udp examples/posix/tailnet_udp.c)
            target_link_libraries(tailnet-udp PRIVATE tailscalec::tailscalec)
        endif()
        if(TSC_APP_TCP)
            add_executable(tailnet-tcp examples/posix/tailnet_tcp.c)
            target_link_libraries(tailnet-tcp PRIVATE tailscalec::tailscalec)
        endif()
    endif()
endif()

if(TSC_BUILD_TESTS)
    include(CTest)
    set(test_modules transport stun)
    if(TSC_NEEDS_CRYPTO)
        list(APPEND test_modules crypto)
    endif()
    if(TSC_NEEDS_STREAM)
        list(APPEND test_modules stream)
    endif()
    if(TSC_CLIENT)
        list(APPEND test_modules packet policy wireguard control_json
            control_map control_protocol control_time control_live client)
    endif()
    if(TSC_DISCO OR TSC_DERP)
        list(APPEND test_modules nacl)
    endif()
    if(TSC_DISCO)
        list(APPEND test_modules disco)
    endif()
    if(TSC_DERP)
        list(APPEND test_modules derp)
    endif()
    if(TSC_APP_TCP)
        list(APPEND test_modules tcp)
    endif()
    foreach(module IN LISTS test_modules)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${module}.c")
            add_executable(test-${module} "tests/test_${module}.c")
            target_link_libraries(test-${module} PRIVATE tailscalec::tailscalec)
            target_include_directories(test-${module} PRIVATE ${TSC_PRIVATE_INCLUDES})
            target_compile_options(test-${module} PRIVATE
                -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
            add_test(NAME ${module} COMMAND test-${module})
            set_tests_properties(${module} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
        endif()
    endforeach()
    if(TSC_CLIENT)
        add_executable(test-wireguard-peer tests/wireguard_peer.c)
        target_link_libraries(test-wireguard-peer PRIVATE tailscalec::tailscalec)
    endif()
    if(TSC_DERP)
        add_executable(test-derp-live tests/interop/derp/client.c)
        target_link_libraries(test-derp-live PRIVATE tailscalec::tailscalec)
    endif()
    if(TSC_DISCO)
        add_executable(test-disco-live tests/interop/disco/client.c)
        target_link_libraries(test-disco-live PRIVATE tailscalec::tailscalec)
    endif()
endif()
